```python
# backend/app/api/knowledge_bases.py (replace in-memory)
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from uuid import UUID

from backend.database import get_db
from backend.app.dependencies import require_platform_user
from backend.app.models.knowledge_base import KnowledgeBase
from backend.app.schemas import KnowledgeBaseCreate, KnowledgeBaseRead, KnowledgeBaseUpdate

router = APIRouter(prefix="/knowledge-bases", tags=["Knowledge Bases"])

@router.post("", response_model=KnowledgeBaseRead, status_code=status.HTTP_201_CREATED)
async def create_kb(payload: KnowledgeBaseCreate, user=Depends(require_platform_user), db=Depends(get_db)):
    kb = KnowledgeBase(tenant_id=user.tenant_id, **payload.model_dump())
    db.add(kb)
    await db.commit()
    await db.refresh(kb)
    return kb

@router.get("", response_model=list[KnowledgeBaseRead])
async def list_kbs(user=Depends(require_platform_user), db=Depends(get_db)):
    result = await db.execute(select(KnowledgeBase).where(KnowledgeBase.tenant_id == user.tenant_id))
    return result.scalars().all()

@router.get("/{kb_id}", response_model=KnowledgeBaseRead)
async def get_kb(kb_id: UUID, user=Depends(require_platform_user), db=Depends(get_db)):
    kb = await db.get(KnowledgeBase, kb_id)
    if not kb or kb.tenant_id != user.tenant_id:
        raise HTTPException(404, "Knowledge base not found")
    return kb

@router.patch("/{kb_id}", response_model=KnowledgeBaseRead)
async def update_kb(kb_id: UUID, payload: KnowledgeBaseUpdate, user=Depends(require_platform_user), db=Depends(get_db)):
    kb = await db.get(KnowledgeBase, kb_id)
    if not kb or kb.tenant_id != user.tenant_id:
        raise HTTPException(404, "Knowledge base not found")
    for k, v in payload.model_dump(exclude_unset=True).items():
        setattr(kb, k, v)
    await db.commit()
    await db.refresh(kb)
    return kb

@router.delete("/{kb_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_kb(kb_id: UUID, user=Depends(require_platform_user), db=Depends(get_db)):
    kb = await db.get(KnowledgeBase, kb_id)
    if not kb or kb.tenant_id != user.tenant_id:
        raise HTTPException(404, "Knowledge base not found")
    await db.delete(kb)
    await db.commit()
```

### 2.4 Ensure Worker Can Access Index

**Problem**: `hybrid_index` is per-process singleton. Worker indexes chunks but API can't see them.

**Solution A (Simple)**: Rehydrate index in API on every search (already in `retrieval_engine.search()`)

**Solution B (Production)**: Use Qdrant as shared vector store
```python
# backend/app/services/indexing/__init__.py - add Qdrant sync
class HybridIndex:
    def __init__(self):
        # ... existing ...
        self._qdrant_client = None
    
    def _get_qdrant(self):
        if self._qdrant_client is None and QDRANT_URL:
            from qdrant_client import QdrantClient
            self._qdrant_client = QdrantClient(url=QDRANT_URL)
        return self._qdrant_client
    
    def index_chunks(self, chunks, embeddings=None):
        # ... existing in-memory indexing ...
        # ALSO write to Qdrant
        qdrant = self._get_qdrant()
        if qdrant:
            self._upsert_qdrant(chunks, embeddings)
    
    def search(self, query, limit=10, ...):
        # Try Qdrant first if available
        qdrant = self._get_qdrant()
        if qdrant:
            return self._search_qdrant(query, limit, ...)
        # Fallback to in-memory
        return self._search_memory(query, limit, ...)
```

## Phase 4: Fix API Key Authentication

### 4.1 Problem
`x-api-key` header waives CSRF but no dependency resolves it to a user.

### 4.2 Solution: Add API Key Dependency

**File: `backend/app/dependencies.py`**

```python
from fastapi import Depends, Header, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
import hmac
import hashlib

from backend.database import get_db
from backend.app.models.api_key import APIKey
from backend.app.models.user import User
from backend.config import API_KEY_HMAC_SECRET

async def get_api_key_user(
    x_api_key: str = Header(None, alias="x-api-key"),
    db: AsyncSession = Depends(get_db),
) -> User:
    """Resolve x-api-key to user."""
    if not x_api_key:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="API key required")
    
    if not x_api_key.startswith("dk_"):
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid API key format")
    
    prefix = x_api_key[:15]  # dk_ + 12 chars
    provided_suffix = x_api_key[15:]
    
    result = await db.execute(select(APIKey).where(APIKey.prefix == prefix))
    api_key = result.scalar_one_or_none()
    
    if not api_key or not api_key.is_active:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid API key")
    
    # Verify HMAC
    expected = hmac.new(
        API_KEY_HMAC_SECRET.encode(),
        provided_suffix.encode(),
        hashlib.sha256
    ).hexdigest()
    
    if not hmac.compare_digest(api_key.key_hash, expected):
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid API key")
    
    # Load user
    result = await db.execute(select(User).where(User.id == api_key.user_id))
    user = result.scalar_one_or_none()
    
    if not user or not user.is_active:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="User not found or disabled")
    
    # Attach for usage logging
    from fastapi import Request
    # This needs request context - see middleware approach below
    
    return user
```

**Alternative: Middleware-based (sets request.state)**


## Phase 8: Cross-Process Index Visibility

### 8.1 Problem
API and worker are separate containers, each with own `hybrid_index` singleton.

### 8.2 Solutions (Pick One)

**Option A: Shared Qdrant (Recommended)**
- Both API and worker write to Qdrant
- Both read from Qdrant
- In-memory index becomes cache only

**Option B: Redis-backed Index**
- Serialize index to Redis
- Both processes read from Redis

**Option C: API-Only Indexing**
- Remove indexing from worker
- API handles all indexing synchronously or via Celery but writes to shared store
- Worker only does non-indexing tasks

**Implementation for Option A:**
```python
# backend/app/services/indexing/__init__.py
class HybridIndex:
    def __init__(self):
        self.keyword_index = KeywordIndex()
        self.vector_index = VectorIndex()
        self.metadata_index = MetadataIndex()
        self.graph_index = GraphIndex()
        self._qdrant = None
    
    def _init_qdrant(self):
        if self._qdrant is None and QDRANT_URL:
            from qdrant_client import QdrantClient
            from qdrant_client.models import Distance, VectorParams
            self._qdrant = QdrantClient(url=QDRANT_URL)
            # Create collection if not exists
            try:
                self._qdrant.get_collection("knowledge_chunks")
            except:
                self._qdrant.create_collection(
                    collection_name="knowledge_chunks",
                    vectors_config=VectorParams(size=1536, distance=Distance.COSINE),
                )
    
    def index_chunks(self, chunks, embeddings=None):
        # In-memory
        entries = list(chunks)
        for entry in entries:
            self.keyword_index.index_chunk(entry)
            self.metadata_index.index_chunk(entry)
        
        chunk_ids = self.vector_index.index_chunks(entries, embeddings)
        
        # Qdrant
        self._init_qdrant()
        if self._qdrant and embeddings:
            from qdrant_client.models import PointStruct
            points = [
                PointStruct(id=entry["chunk_id"], vector=emb, payload=entry)
                for entry, emb in zip(entries, embeddings)
            ]
            self._qdrant.upsert(collection_name="knowledge_chunks", points=points)
        
        return chunk_ids
    
    def search(self, query, limit=10, metadata_filter=None, kb_ids=None):
        self._init_qdrant()
        if self._qdrant:
            return self._search_qdrant(query, limit, metadata_filter, kb_ids)
        return self._search_memory(query, limit, metadata_filter, kb_ids)
```
